# 1. Create a lightweight isolated Python sandbox
    python3 -m venv .venv

# 2. Activate that sandbox
    source .venv/bin/activate  # Windows: .venv\Scripts\activate

# 3. Install dependencies inside the sandbox
    pip install -r requirements.txt
